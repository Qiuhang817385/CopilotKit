适用于 OpenAI 的 Copilot Runtime 适配器。
 
## 示例
 
```ts
import { CopilotRuntime, OpenAIAdapter } from "@copilotkit/runtime";
import OpenAI from "openai";
 
const copilotKit = new CopilotRuntime();
 
const openai = new OpenAI({
  organization: "<your-organization-id>", // 可选
  apiKey: "<your-api-key>",
});
 
return new OpenAIAdapter({ openai });
```
 
## 使用 Azure OpenAI 的示例
 
```ts
import { CopilotRuntime, OpenAIAdapter } from "@copilotkit/runtime";
import OpenAI from "openai";
 
// 您的 Azure OpenAI 实例名称。
// https://learn.microsoft.com/en-us/azure/cognitive-services/openai/how-to/create-resource?pivots=web-portal#create-a-resource
const instance = "<your instance name>";
 
// 对应于您的 OpenAI 资源中的模型部署，例如 my-gpt35-16k-deployment
// 导航到 Azure OpenAI Studio 以部署模型。
const model = "<your model>";
 
const apiKey = process.env["AZURE_OPENAI_API_KEY"];
if (!apiKey) {
  throw new Error("AZURE_OPENAI_API_KEY 环境变量缺失或为空。");
}
 
const copilotKit = new CopilotRuntime();
 
const openai = new OpenAI({
  apiKey,
  baseURL: `https://${instance}.openai.azure.com/openai/deployments/${model}`,
  defaultQuery: { "api-version": "2024-04-01-preview" },
  defaultHeaders: { "api-key": apiKey },
});
 
return new OpenAIAdapter({ openai });
```

## 构造函数参数

<PropertyReference name="openai" type="OpenAI"  > 
要使用的可选 OpenAI 实例。如果未提供，将创建一个新实例。
</PropertyReference>

<PropertyReference name="model" type="string"  > 
要使用的模型。
</PropertyReference>

<PropertyReference name="disableParallelToolCalls" type="boolean"  default="false"> 
是否禁用并行工具调用。
  您可以禁用并行工具调用以强制模型按顺序执行工具调用。
  当您希望按特定顺序执行工具调用，以便一个工具调用引入的状态更改对下一个工具调用可见时，这很有用。（即新的 actions 或 readables）
</PropertyReference>

<PropertyReference name="keepSystemRole" type="boolean"  default="false"> 
是否在系统消息中保持角色为 "System"。
  默认情况下，它会被转换为 "developer"，这是较新的 OpenAI 模型所使用的
</PropertyReference>
