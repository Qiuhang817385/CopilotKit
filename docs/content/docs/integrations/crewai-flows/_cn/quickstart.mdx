## 先决条件

在开始之前，您必须已经在 [CrewAI Enterprise](https://docs.crewai.com/enterprise/introduction) 上部署了 [CrewAI Flow](https://docs.crewai.com/guides/flows/first-flow)。如果您正在寻找示例流程，请查看这个[示例代理聊天实现](https://github.com/suhasdeshpande/agentic_chat)。

            ### 创建 CrewAI Flow 组件
            将以下代码片段放在您的**主页面**（例如 Next.js 中的 `page.tsx`）或您想使用 CopilotKit 的任何地方。

            ```tsx title="page.tsx"
            "use client";
            import "@copilotkit/react-ui/v2/styles.css";
            import { CopilotKit } from "@copilotkit/react-core";
            import { z } from "zod";
            import { useFrontendTool } from "@copilotkit/react-core/v2";
            import { CopilotChat } from "@copilotkit/react-core/v2";
            import React, { useState } from "react";

            const publicApiKey = process.env.NEXT_PUBLIC_COPILOT_API_KEY || "";
            /**
             * AgentName 指的是您在设置期间通过 CLI 保存的 Crew Flow 代理。
             * 它用于标识您要用于聊天的代理。
             */
            const agentName =
              process.env.NEXT_PUBLIC_COPILOTKIT_AGENT_NAME || "DefaultAgent";

            // 主聊天组件：处理聊天界面和背景自定义
            const Chat = () => {
              const [background, setBackground] = useState(
                "linear-gradient(135deg, #667eea 0%, #764ba2 100%)"
              );

              // 操作：允许 AI 动态更改背景颜色
              useFrontendTool({
                name: "change_background",
                description:
                  "更改聊天的背景颜色。可以是 CSS background 属性接受的任何值。常规颜色、线性或径向渐变等。",
                parameters: z.object({
                  background: z.string().describe("背景。优先使用渐变。"),
                }),
                handler: async ({ background }) => {
                  setBackground(background);
                  return `背景已更改为 ${background}`;
                },
                followUp: false,
              });

              return (
                <div
                  className="h-screen w-full flex items-center justify-center"
                  style={{ background }}
                >
                  <div className="w-full max-w-3xl h-[80vh] px-4">
                    <div className="h-full bg-white/10 backdrop-blur-lg rounded-2xl shadow-2xl overflow-hidden">
                      <CopilotChat
                        className="h-full"
                        labels={{
                          welcomeMessageText: "嗨，我是一个代理。想聊天吗？",
                          placeholder: "输入消息...",
                        }}
                      />
                    </div>
                  </div>
                </div>
              );
            }

            // App 组件：提供 CopilotKit 上下文的主包装器
            const CrewAIFlow: React.FC = () => (
              <CopilotKit publicApiKey={publicApiKey} agent={agentName}>
                <Chat />
              </CopilotKit>
            );

            export default CrewAIFlow;
            ```
