## 先决条件

在开始之前，您需要以下内容：

- OpenAI API 密钥
- Node.js 20+
- Python 3.12+
- 您喜欢的包管理器

            <Step>
                ### 初始化代理项目

                如果您还没有 Python 项目，请使用 `uv` 创建一个：

                ```bash
                uv init my-agent
                cd my-agent
                ```
            </Step>
            <Step>
                ### 安装带 AG-UI 的 Strands

                将 Strands 和所需包添加到您的项目：

                ```bash
                uv add ag-ui-strands "strands-agents[OpenAI]" fastapi uvicorn
                ```
            </Step>
            <Step>
                ### 配置环境

                将您的 OpenAI API 密钥设置为环境变量：

                ```bash
                export OPENAI_API_KEY=your_openai_api_key
                ```

                <Callout type="info" title="其他模型呢？">
                  本示例使用 OpenAI 的 GPT-4o，但您可以修改它以使用 Strands 支持的任何语言模型。
                </Callout>
            </Step>
            <Step>
                ### 通过 AG-UI 暴露您的代理

                更新代理文件以将其作为 AG-UI ASGI 应用程序暴露：

                ```python title="main.py"
                import os

                from ag_ui_strands import StrandsAgent, create_strands_app
                from strands import Agent
                from strands.models.openai import OpenAIModel

                # 设置您的 Strands 代理
                api_key = os.getenv("OPENAI_API_KEY", "")
                model = OpenAIModel(
                    client_args={"api_key": api_key},
                    model_id="gpt-5.2",
                )

                agent = Agent(
                    model=model,
                    system_prompt="您是一个乐于助人的 AI 助手。",
                )

                # 包装 AG-UI 集成
                agui_agent = StrandsAgent(
                    agent=agent,
                    name="strands_agent",
                )

                # 创建 FastAPI 应用
                app = create_strands_app(agui_agent, "/")

                if __name__ == "__main__":
                    import uvicorn

                    uvicorn.run("main:app", host="0.0.0.0", port=8000, reload=True)

                ```

                <Callout type="info" title="什么是 AG-UI？">
                  AG-UI 是一个用于前端-代理通信的开放协议。`create_strands_app` 函数创建一个 CopilotKit 可以连接的 ASGI 应用。
                </Callout>
            </Step>
            <Step>
                ### 创建前端

                CopilotKit 适用于任何基于 React 的前端。本示例使用 Next.js。

                ```bash
                npx create-next-app@latest frontend
                cd frontend
                ```
            </Step>
            <Step>
                ### 安装 CopilotKit 包

                ```npm
                npm install @copilotkit/react-ui @copilotkit/react-core @copilotkit/runtime @ag-ui/client
                ```
            </Step>
            <Step>
                ### 设置 Copilot Runtime

                创建一个 API 路由将 CopilotKit 连接到您的 Strands 代理：

                ```tsx title="app/api/copilotkit/route.ts"
                import {
                  CopilotRuntime,
                  ExperimentalEmptyAdapter,
                  copilotRuntimeNextJSAppRouterEndpoint,
                } from "@copilotkit/runtime";
                import { HttpAgent } from "@ag-ui/client";
                import { NextRequest } from "next/server";

                const serviceAdapter = new ExperimentalEmptyAdapter();

                const runtime = new CopilotRuntime({
                  agents: {
                    strands_agent: new HttpAgent({ url: "http://localhost:8000" }),
                  }
                });

                export const POST = async (req: NextRequest) => {
                  const { handleRequest } = copilotRuntimeNextJSAppRouterEndpoint({
                    runtime,
                    serviceAdapter,
                    endpoint: "/api/copilotkit",
                  });

                  return handleRequest(req);
                };
               ```
            </Step>
            <Step>
                ### 配置 CopilotKit Provider

                使用 CopilotKit provider 包装您的应用程序：

                ```tsx title="app/layout.tsx"
                import { CopilotKit } from "@copilotkit/react-core"; // [!code highlight]
                import "@copilotkit/react-ui/v2/styles.css";

                // ...

                export default function RootLayout({ children }: {children: React.ReactNode}) {
                  return (
                    <html lang="en">
                      <body>
                        {/* [!code highlight:3] */}
                        <CopilotKit runtimeUrl="/api/copilotkit" agent="strands_agent">
                          {children}
                        </CopilotKit>
                      </body>
                    </html>
                  );
                }
                ```
            </Step>
            <Step>
              ### 添加聊天界面

              将 CopilotSidebar 组件添加到您的页面：

              ```tsx title="app/page.tsx"
              import { CopilotSidebar } from "@copilotkit/react-core/v2"; // [!code highlight:1]

              export default function Page() {
                return (
                  <main>
                    <h1>您的应用</h1>
                    {/* [!code highlight:1] */}
                    <CopilotSidebar />
                  </main>
                );
              }
              ```
            </Step>
            <Step>
                ### 启动代理

                从代理目录启动代理服务器：

                ```bash
                cd ..
                uv run main.py
                ```

                您的代理将在 `http://localhost:8000` 上可用。
            </Step>
            <Step>
