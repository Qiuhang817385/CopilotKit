## 先决条件

在开始之前，您需要以下内容：

- OpenAI API 密钥
- Node.js 20+
- 您喜欢的包管理器

            id="bring-your-own"
            title="使用现有代理"
            description="我已经有一个 Mastra 代理，想要添加 CopilotKit。"
        >
            <Step>
                ### 初始化您的 Mastra 项目

                如果您还没有 Mastra 项目设置，请创建一个：

                ```bash
                npx create-mastra@latest my-agent
                cd my-agent
                ```
            </Step>
            <Step>
                ### 创建您的 Mastra 代理

                在您的 Mastra 项目中创建一个新的代理文件：

                ```ts title="src/mastra/agents/index.ts"
                import { openai } from "@ai-sdk/openai";
                import { Agent } from "@mastra/core/agent";

                export const myAgent = new Agent({
                  name: "My Agent",
                  instructions: "You are a helpful assistant!",
                  model: openai("gpt-5.2"),
                });
                ```

                然后从您的 Mastra 实例导出它：

                ```ts title="src/mastra/index.ts"
                import { Mastra } from "@mastra/core";
                import { myAgent } from "./agents";

                export const mastra = new Mastra({
                  agents: { myAgent },
                });
                ```

                <Callout type="info" title="其他模型呢？">
                  此示例使用 OpenAI 的 GPT-4o，但您可以修改它以使用 Mastra 支持的任何语言模型。
                </Callout>
            </Step>
            <Step>
                ### 配置您的环境

                将您的 OpenAI API 密钥设置为环境变量：

                ```bash
                export OPENAI_API_KEY=your_openai_api_key
                ```
            </Step>
            <Step>
                ### 创建您的前端

                CopilotKit 适用于任何基于 React 的前端。在此示例中，我们将使用 Next.js。

                ```bash
                npx create-next-app@latest my-copilot-app
                cd my-copilot-app
                ```
            </Step>
            <Step>
                ### 安装 CopilotKit 包

                ```npm
                npm install @copilotkit/react-ui @copilotkit/react-core @copilotkit/runtime @ag-ui/mastra @ag-ui/core @ag-ui/client @mastra/client-js @ai-sdk/openai
                ```
            </Step>
            <Step>
                ### 设置 Copilot Runtime

                创建一个 API 路由，将 CopilotKit 连接到您的 Mastra 代理：

                ```ts title="app/api/copilotkit/route.ts"
                import {
                  CopilotRuntime,
                  ExperimentalEmptyAdapter,
                  copilotRuntimeNextJSAppRouterEndpoint,
                } from "@copilotkit/runtime";
                import { NextRequest } from "next/server";
                import { MastraAgent } from "@ag-ui/mastra"
                import { mastra } from "@/mastra"; // 指向您的 Mastra 实例的路径

                const serviceAdapter = new ExperimentalEmptyAdapter();

                const runtime = new CopilotRuntime({
                  agents: MastraAgent.getLocalAgents({ mastra }),
                });

                export const POST = async (req: NextRequest) => {
                  const { handleRequest } = copilotRuntimeNextJSAppRouterEndpoint({
                    runtime,
                    serviceAdapter,
                    endpoint: "/api/copilotkit",
                  });

                  return handleRequest(req);
                };
                ```
            </Step>
            <Step>
                ### 配置 CopilotKit Provider

                使用 CopilotKit Provider 包装您的应用程序：

                ```tsx title="app/layout.tsx"
                import { CopilotKit } from "@copilotkit/react-core"; // [!code highlight]
                import "@copilotkit/react-ui/v2/styles.css";

                // ...

                export default function RootLayout({ children }: {children: React.ReactNode}) {
                  return (
                    <html lang="en">
                      <body>
                        {/* [!code highlight:3] */}
                        <CopilotKit runtimeUrl="/api/copilotkit" agent="myAgent">
                          {children}
                        </CopilotKit>
                      </body>
                    </html>
                  );
                }
                ```
            </Step>
            <Step>
              ### 添加聊天界面

              将 CopilotSidebar 组件添加到您的页面：

              ```tsx title="app/page.tsx"
              import { CopilotSidebar } from "@copilotkit/react-core/v2"; // [!code highlight:1]

              export default function Page() {
                return (
                  <main>
                    <h1>Your App</h1>
                    {/* [!code highlight:1] */}
                    <CopilotSidebar />
                  </main>
                );
              }
              ```
            </Step>
            <Step>
                ### 启动您的 UI

                启动开发服务器：

                <Tabs groupId="package-manager" items={['npm', 'pnpm', 'yarn', 'bun']}>
                    <Tab value="npm">
                        ```bash
                        npm run dev
                        ```
                    </Tab>
                    <Tab value="pnpm">
                        ```bash
                        pnpm dev
                        ```
                    </Tab>
                    <Tab value="yarn">
                        ```bash
                        yarn dev
                        ```
                    </Tab>
                    <Tab value="bun">
                        ```bash
                        bun dev
                        ```
                    </Tab>
                </Tabs>
            </Step>
        </TailoredContentOption>
