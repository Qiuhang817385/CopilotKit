## 这是什么？

无头 UI 让您完全控制聊天体验 —— 您提供自己的组件、布局和样式，而 CopilotKit 处理 Agent 通信、消息管理和流式传输。这建立在 [程序化控制](/langgraph/programmatic-control) 中介绍的相同原语（`useAgent` 和 `useCopilotKit`）之上。

## 何时使用？

当 [插槽系统](/langgraph/custom-look-and-feel/slots) 不够用时使用无头 UI —— 例如，当您需要完全不同的布局、想要将聊天嵌入到现有 UI 中，或者正在构建非聊天界面但仍需要与 Agent 通信时。

## 实现

<Steps>
<Step>
### 访问 Agent 和 CopilotKit

使用 `useAgent` 获取 Agent 实例（消息、状态、执行状态）和 `useCopilotKit` 来运行 Agent。

```tsx title="components/custom-chat.tsx"
import { useAgent } from "@copilotkit/react-core/v2";
import { useCopilotKit } from "@copilotkit/react-core/v2";
import { randomUUID } from "@copilotkit/shared/v2";

export function CustomChat() {
  // [!code highlight:2]
  const { agent } = useAgent();
  const { copilotkit } = useCopilotKit();

  return <div>{/* 您的自定义 UI */}</div>;
}
```
</Step>

<Step>
### 显示消息

Agent 的消息可通过 `agent.messages` 获取。每条消息都有 `id`、`role`（`"user"` 或 `"assistant"`）和 `content`。

```tsx title="components/custom-chat.tsx"
export function CustomChat() {
  const { agent } = useAgent();
  const { copilotkit } = useCopilotKit();

  return (
    <div className="flex flex-col h-full">
      {/* [!code highlight:12] */}
      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {agent.messages.map((msg) => (
          <div
            key={msg.id}
            className={msg.role === "user" ? "ml-auto bg-blue-100 rounded-lg p-3 max-w-md" : "bg-gray-100 rounded-lg p-3 max-w-md"}
          >
            <p className="text-sm font-medium">{msg.role}</p>
            <p>{msg.content}</p>
          </div>
        ))}
        {agent.isRunning && <div className="text-gray-400">思考中...</div>}
      </div>
    </div>
  );
}
```
</Step>

<Step>
### 发送消息并运行 Agent

将消息添加到 Agent 的对话中，然后调用 `copilotkit.runAgent()` 触发执行。这与 CopilotKit 内置的 `<CopilotChat />` 内部使用的方法相同。

```tsx title="components/custom-chat.tsx"
import { useState, useCallback } from "react";

export function CustomChat() {
  const { agent } = useAgent();
  const { copilotkit } = useCopilotKit();
  const [input, setInput] = useState("");

  // [!code highlight:14]
  const sendMessage = useCallback(async () => {
    if (!input.trim()) return;

    agent.addMessage({
      id: randomUUID(),
      role: "user",
      content: input,
    });

    setInput("");

    await copilotkit.runAgent({ agent });
  }, [input, agent, copilotkit]);

  return (
    <div className="flex flex-col h-full">
      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {agent.messages.map((msg) => (
          <div key={msg.id} className={msg.role === "user" ? "ml-auto bg-blue-100 rounded-lg p-3 max-w-md" : "bg-gray-100 rounded-lg p-3 max-w-md"}>
            <p>{msg.content}</p>
          </div>
        ))}
        {agent.isRunning && <div className="text-gray-400">思考中...</div>}
      </div>

      {/* [!code highlight:12] */}
      <form
        className="border-t p-4 flex gap-2"
        onSubmit={(e) => { e.preventDefault(); sendMessage(); }}
      >
        <input
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder="输入消息..."
          className="flex-1 border rounded-lg px-3 py-2"
        />
        <button type="submit" disabled={agent.isRunning}>发送</button>
      </form>
    </div>
  );
}
```
</Step>

<Step>
### 停止 Agent

使用 `copilotkit.stopAgent()` 取消正在运行的 Agent：

```tsx title="components/custom-chat.tsx"
const stopAgent = useCallback(() => {
  // [!code highlight:1]
  copilotkit.stopAgent({ agent });
}, [agent, copilotkit]);

// 在您的 JSX 中：
{agent.isRunning && (
  <button onClick={stopAgent} className="text-red-500">
    停止
  </button>
)}
```
</Step>

<Step>
### 订阅 Agent 事件

使用 `agent.subscribe()` 监听生命周期事件 —— 对于显示进度指示器、处理错误或响应自定义事件（如 LangGraph 中断）很有用。

```tsx title="components/custom-chat.tsx"
import { useEffect, useState } from "react";
import type { AgentSubscriber } from "@ag-ui/client";

export function CustomChat() {
  const { agent } = useAgent();
  const { copilotkit } = useCopilotKit();
  const [interrupt, setInterrupt] = useState<string | null>(null);

  // [!code highlight:16]
  useEffect(() => {
    const subscriber: AgentSubscriber = {
      onCustomEvent: ({ event }) => {
        if (event.name === "on_interrupt") {
          setInterrupt(event.value);
        }
      },
    };

    const { unsubscribe } = agent.subscribe(subscriber);
    return () => unsubscribe();
  }, [agent]);

  const resolveInterrupt = (response: string) => {
    agent.runAgent({
      forwardedProps: { command: { resume: response } },
    });
    setInterrupt(null);
  };

  return (
    <div>
      {/* 消息和输入... */}

      {interrupt && (
        <div className="fixed inset-0 bg-black/50 flex items-center justify-center">
          <div className="bg-white rounded-xl p-6 max-w-md">
            <p className="font-medium mb-4">{interrupt}</p>
            <form onSubmit={(e) => {
              e.preventDefault();
              const formData = new FormData(e.currentTarget);
              resolveInterrupt(formData.get("response") as string);
            }}>
              <input name="response" className="border rounded px-3 py-2 w-full mb-3" />
              <button type="submit" className="bg-blue-500 text-white px-4 py-2 rounded">
                提交
              </button>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
```
</Step>

<Step>
### 访问共享状态

如果您的 LangGraph Agent 与前端共享状态，可以通过 `agent.state` 访问它：

```tsx title="components/custom-chat.tsx"
export function AgentDashboard() {
  const { agent } = useAgent();

  // [!code highlight:3]
  const currentNode = agent.state.currentNode;
  const progress = agent.state.progress;
  const results = agent.state.results;

  return (
    <div>
      {currentNode && <div className="text-sm text-gray-500">当前步骤：{currentNode}</div>}
      {progress && <div className="w-full bg-gray-200 rounded"><div className="bg-blue-500 h-2 rounded" style={{ width: `${progress}%` }} /></div>}
      {results && <pre className="bg-gray-50 p-4 rounded">{JSON.stringify(results, null, 2)}</pre>}
    </div>
  );
}
```
</Step>
</Steps>
