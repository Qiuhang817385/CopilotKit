## 准备工作

在开始之前，你需要准备以下内容：

- OpenAI API 密钥
- Node.js 20+
- 你喜欢的包管理器
- （可选）LangSmith API 密钥 - 仅在使用现有 LangGraph 智能体时需要

            ### 通过 AG-UI 暴露你的智能体

            如果你已经编写了一个 LangGraph 智能体，请参考以下代码。在此步骤中，我们创建一个简单的 LangGraph 智能体用于演示。
              <Tabs groupId="deployment_method" items={['LangSmith', 'FastAPI']}>
                <Tab value="LangSmith">
                  首先，我们将创建一个简单的 LangGraph 智能体：

                  ```python title="main.py"
                  from langchain_core.messages import SystemMessage
                  from langchain_openai import ChatOpenAI
                  from langgraph.graph import END, START, MessagesState, StateGraph

                  async def mock_llm(state: MessagesState):
                    model = ChatOpenAI(model="gpt-4.1-mini")
                    system_message = SystemMessage(content="你是一个有用的助手。")
                    response = await model.ainvoke(
                      [
                        system_message,
                        *state["messages"],
                      ]
                    )
                    return {"messages": response}

                  graph = StateGraph(MessagesState)
                  graph.add_node(mock_llm)
                  graph.add_edge(START, "mock_llm")
                  graph.add_edge("mock_llm", END)
                  graph = graph.compile()
                  ```

                  然后为了使用 LangSmith 进行测试和部署，我们还需要一个 `langgraph.json`

                  ```sh
                  touch langgraph.json
                  ```

                  ```json title="langgraph.json"
                  {
                    "python_version": "3.12",
                    "dockerfile_lines": [],
                    "dependencies": ["."],
                    "package_manager": "uv",
                    "graphs": {
                      "sample_agent": "./main.py:graph"
                    },
                    "env": ".env"
                  }
                  ```
                </Tab>
                <Tab value="FastAPI">
                  首先，将 `ag-ui-langgraph` 包添加到你的项目中：

                  ```bash
                  uv add ag-ui-langgraph fastapi uvicorn copilotkit
                  ```

                  然后创建一个简单的 LangGraph 智能体，添加 FastAPI 应用，并将我们的智能体作为 AG-UI 端点附加。

                  ```python title="main.py"
                  import os

                  # [!code highlight:2]
                  from ag_ui_langgraph import add_langgraph_fastapi_endpoint
                  from copilotkit import LangGraphAGUIAgent
                  from fastapi import FastAPI
                  from langgraph.graph import END, START, MessagesState, StateGraph
                  from langchain_core.messages import SystemMessage
                  from langchain_openai import ChatOpenAI

                  async def mock_llm(state: MessagesState):
                    model = ChatOpenAI(model="gpt-4.1-mini")
                    system_message = SystemMessage(content="你是一个有用的助手。")
                    response = await model.ainvoke(
                      [
                        system_message,
                        *state["messages"],
                      ]
                    )
                    return {"messages": response}


                  graph = StateGraph(MessagesState)
                  graph.add_node(mock_llm)
                  graph.add_edge(START, "mock_llm")
                  graph.add_edge("mock_llm", END)
                  graph = graph.compile()

                  app = FastAPI()

                  # [!code highlight:9]
                  add_langgraph_fastapi_endpoint(
                    app=app,
                    agent=LangGraphAGUIAgent(
                      name="sample_agent",
                      description="一个示例智能体，可用作你自己智能体的起点。",
                      graph=graph,
                    ),
                    path="/",
                  )

                  def main():
                    """运行 uvicorn 服务器。"""
                    uvicorn.run(
                      "main:app",
                      host="0.0.0.0",
                      port="8123",
                      reload=True,
                    )

                  if __name__ == "__main__":
                    main()
                  ```

              ### 设置 Copilot Runtime

              创建一个 API 路由将 CopilotKit 连接到你的 LangGraph 智能体：

              ```sh
              mkdir -p app/api/copilotkit && touch app/api/copilotkit/route.ts
              ```

              <Tabs groupId="deployment_method" items={['LangSmith', 'FastAPI']}>
                <Tab value="LangSmith">
                  ```tsx title="app/api/copilotkit/route.ts"
                  import {
                    CopilotRuntime,
                    ExperimentalEmptyAdapter,
                    copilotRuntimeNextJSAppRouterEndpoint,
                  } from "@copilotkit/runtime";
                  // [!code highlight]
                  import { LangGraphAgent } from "@copilotkit/runtime/langgraph";
                  import { NextRequest } from "next/server";

                  const serviceAdapter = new ExperimentalEmptyAdapter();

                  const runtime = new CopilotRuntime({
                    agents: {
                    // [!code highlight:5]
                      sample_agent: new LangGraphAgent({
                        deploymentUrl:  process.env.LANGGRAPH_DEPLOYMENT_URL || "http://localhost:8123",
                        graphId: "sample_agent",
                        langsmithApiKey: process.env.LANGSMITH_API_KEY || "",
                      }),
                    }
                  });

                  export const POST = async (req: NextRequest) => {
                    const { handleRequest } = copilotRuntimeNextJSAppRouterEndpoint({
                      runtime,
                      serviceAdapter,
                      endpoint: "/api/copilotkit",
                    });

                    return handleRequest(req);
                  };
                ```
              </Tab>
              <Tab value="FastAPI">
                ```tsx title="app/api/copilotkit/route.ts"
                import {
                  CopilotRuntime,
                  ExperimentalEmptyAdapter,
                  copilotRuntimeNextJSAppRouterEndpoint,
                } from "@copilotkit/runtime";
                // [!code highlight]
                import { LangGraphHttpAgent } from "@copilotkit/runtime/langgraph";
                import { NextRequest } from "next/server";

                const serviceAdapter = new ExperimentalEmptyAdapter();

                const runtime = new CopilotRuntime({
                  agents: {
                    // [!code highlight:3]
                    sample_agent: new LangGraphHttpAgent({
                      url:  process.env.LANGGRAPH_DEPLOYMENT_URL || "http://localhost:8123",
                    }),
                  }
                });

                export const POST = async (req: NextRequest) => {
                  const { handleRequest } = copilotRuntimeNextJSAppRouterEndpoint({
                    runtime,
                    serviceAdapter,
                    endpoint: "/api/copilotkit",
                  });

                  return handleRequest(req);
                };
              ```

              使用 CopilotKit provider 包装你的应用程序：

              import "@copilotkit/react-ui/v2/styles.css";

              // ...

              export default function RootLayout({ children }: {children: React.ReactNode}) {
                return (
                  <html lang="zh">
                    <body>
                      {/* [!code highlight:3] */}
                      <CopilotKit runtimeUrl="/api/copilotkit" agent="sample_agent">

                      </CopilotKit>
